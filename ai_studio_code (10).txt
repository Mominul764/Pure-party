import React, { useMemo, useState } from 'react';
import { Lock, Plus, Rocket, Search, Sparkles } from 'lucide-react';
import { RECHARGE_PACKAGES } from '../data/economySpecs';
import { EconomyConfig, GiftTier, UserProfile } from '../types/architecture';
import {
  calculateGiftSplit,
  canCreateCustom1MGift,
  formatCoins,
  formatMillionsLabel,
} from '../utils/economyMath';

interface GiftAndCustomStudioProps {
  user: UserProfile;
  economy: EconomyConfig;
  gifts: GiftTier[];
  onCreateCustomGift: (gift: GiftTier) => void;
  onSendGiftToBoat: (gift: GiftTier, catcherLimit: number) => void;
  onRechargePackage: (totalCoins: number, label: string, usdPrice: number) => void;
}

type FilterBracket =
  | 'all'
  | '1-25'
  | '26-50'
  | '51-75'
  | '76-100'
  | 'rockets'
  | 'luckybacks'
  | 'custom';

export const GiftAndCustomStudio: React.FC<GiftAndCustomStudioProps> = ({
  user,
  economy,
  gifts,
  onCreateCustomGift,
  onSendGiftToBoat,
  onRechargePackage,
}) => {
  const [searchQuery, setSearchQuery] = useState('');
  const [bracket, setBracket] = useState<FilterBracket>('all');

  const [customName, setCustomName] = useState('Sovereign Harbor Armada');
  const [customCoins, setCustomCoins] = useState(1_000_000);
  const [customIsRocket, setCustomIsRocket] = useState(true);
  const [customIsLuckyBack, setCustomIsLuckyBack] = useState(true);
  const [customCatchers, setCustomCatchers] = useState(30);
  const [createdBanner, setCreatedBanner] = useState<string | null>(null);

  const eligibleForCustom1M = canCreateCustom1MGift(user);

  const filteredGifts = useMemo(() => {
    return gifts.filter((g) => {
      if (
        searchQuery.trim() &&
        !g.name.toLowerCase().includes(searchQuery.toLowerCase()) &&
        !String(g.tier).includes(searchQuery.trim())
      ) {
        return false;
      }
      if (bracket === '1-25') return g.tier >= 1 && g.tier <= 25;
      if (bracket === '26-50') return g.tier >= 26 && g.tier <= 50;
      if (bracket === '51-75') return g.tier >= 51 && g.tier <= 75;
      if (bracket === '76-100') return g.tier >= 76 && g.tier <= 100;
      if (bracket === 'rockets') return g.isRocket;
      if (bracket === 'luckybacks') return g.isLuckyBack;
      if (bracket === 'custom') return g.category === 'Custom';
      return true;
    });
  }, [gifts, searchQuery, bracket]);

  const handleCreateCustom = (e: React.FormEvent) => {
    e.preventDefault();
    if (!eligibleForCustom1M) return;
    const validCoins = Math.max(1_000_000, Math.round(customCoins));
    const newTierNum = gifts.length + 1;
    const newGift: GiftTier = {
      id: `gift-custom-${Date.now()}`,
      tier: newTierNum,
      name: customName.trim() || `Custom Gift #${newTierNum}`,
      category: 'Custom',
      coinCost: validCoins,
      isRocket: customIsRocket,
      isLuckyBack: customIsLuckyBack,
      defaultCatcherLimit: Math.min(50, Math.max(10, customCatchers)),
      xpYield: Math.round(validCoins * 0.5),
      spYield: Math.round((validCoins * economy.seatSpSplitPct) / 100),
      creatorName: user.name,
    };
    onCreateCustomGift(newGift);
    setCreatedBanner(
      `Published "${newGift.name}" (${formatCoins(newGift.coinCost)} coins) to catalog and Boat Room.`
    );
  };

  return (
    <div className="space-y-8">
      {/* Top Split: Custom 1M Gift Studio + PURE 10x Recharge Cards */}
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-6 items-stretch">
        {/* Custom 1 Million Coin Gift Creator (SVIP 4+ or Level 150+) */}
        <section className="lg:col-span-6 bg-white border border-slate-200 rounded-xl p-6 flex flex-col justify-between space-y-5">
          <div className="space-y-2">
            <div className="flex items-center justify-between gap-3">
              <h2 className="font-display text-lg font-semibold text-slate-900">
                Custom 1 Million Coin Gift Studio
              </h2>
              <span
                className={`text-xs font-medium font-mono tabular-nums ${
                  eligibleForCustom1M ? 'text-emerald-700' : 'text-red-700'
                }`}
              >
                {eligibleForCustom1M
                  ? `Eligible · SVP ${user.svpLevel} / Lv.${user.level}`
                  : `Locked · Need SVP 4+ or Lv.150+`}
              </span>
            </div>
            <p className="text-xs text-slate-600 leading-relaxed">
              Exclusive custom gift authoring unlocked at SVP Level 4+ or Account Level 150+. Minimum valuation is 1,000,000 coins with optional Rocket launch trajectory and 10–50 Lucky Pool catcher slots.
            </p>
          </div>

          {!eligibleForCustom1M ? (
            <div className="bg-slate-50 border border-slate-200 rounded-lg p-5 space-y-2">
              <div className="flex items-center gap-2 text-sm font-semibold text-slate-900">
                <Lock className="w-4 h-4 text-red-600" />
                Studio Locked for Current Profile
              </div>
              <p className="text-xs text-slate-600">
                Your active profile is Level {user.level} and SVP {user.svpLevel}. Switch to a higher-tier profile preset in the session bar above or upgrade via Super Admin controls to unlock Custom 1M Coin Gift creation.
              </p>
            </div>
          ) : (
            <form onSubmit={handleCreateCustom} className="space-y-4">
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-medium text-slate-700 mb-1">
                    Custom Gift Name
                  </label>
                  <input
                    type="text"
                    required
                    value={customName}
                    onChange={(e) => setCustomName(e.target.value)}
                    className="w-full px-3 py-2 text-xs bg-white border border-slate-200 rounded-lg focus:outline-none focus:border-blue-600"
                  />
                </div>
                <div>
                  <label className="block text-xs font-medium text-slate-700 mb-1">
                    Coin Value (Min 1,000,000)
                  </label>
                  <input
                    type="number"
                    min={1_000_000}
                    max={50_000_000}
                    step={100_000}
                    value={customCoins}
                    onChange={(e) => setCustomCoins(Number(e.target.value))}
                    className="w-full px-3 py-2 text-xs font-mono tabular-nums bg-white border border-slate-200 rounded-lg focus:outline-none focus:border-blue-600"
                  />
                </div>
              </div>

              <div className="flex flex-wrap items-center gap-5 text-xs text-slate-700">
                <label className="inline-flex items-center gap-2 cursor-pointer">
                  <input
                    type="checkbox"
                    checked={customIsRocket}
                    onChange={(e) => setCustomIsRocket(e.target.checked)}
                    className="accent-blue-600"
                  />
                  Rocket Trajectory
                </label>
                <label className="inline-flex items-center gap-2 cursor-pointer">
                  <input
                    type="checkbox"
                    checked={customIsLuckyBack}
                    onChange={(e) => setCustomIsLuckyBack(e.target.checked)}
                    className="accent-blue-600"
                  />
                  Enable {economy.luckyPoolPct}% Lucky Pool
                </label>
                <label className="inline-flex items-center gap-2">
                  <span>Catchers:</span>
                  <input
                    type="number"
                    min={10}
                    max={50}
                    value={customCatchers}
                    onChange={(e) => setCustomCatchers(Number(e.target.value))}
                    className="w-16 px-2 py-1 font-mono tabular-nums border border-slate-200 rounded-md"
                  />
                </label>
              </div>

              {createdBanner && (
                <div className="text-xs text-emerald-700 bg-emerald-50 border border-emerald-200 rounded-lg px-3 py-2">
                  {createdBanner}
                </div>
              )}

              <button
                type="submit"
                className="inline-flex items-center gap-1.5 px-4 py-2 text-xs font-semibold text-white bg-slate-900 hover:bg-slate-800 rounded-lg transition-colors cursor-pointer"
              >
                <Plus className="w-3.5 h-3.5" />
                Publish Custom 1M+ Gift
              </button>
            </form>
          )}
        </section>

        {/* PURE Branded 10x Recharge Cards (1M to 100,000M) */}
        <section className="lg:col-span-6 bg-white border border-slate-200 rounded-xl p-6 space-y-4">
          <div className="flex items-center justify-between gap-3">
            <div>
              <h2 className="font-display text-lg font-semibold text-slate-900">
                PURE Branded 10x Recharge Cards
              </h2>
              <p className="text-xs text-slate-600 mt-0.5">
                PURE denominations from 1M to 100,000M coins guaranteeing a 10x return toward the $20,000 ceiling.
              </p>
            </div>
            <div className="text-right font-mono tabular-nums">
              <div className="text-[11px] text-slate-500">Current Balance</div>
              <div className="text-sm font-semibold text-slate-900">
                {formatCoins(user.coinBalance)}
              </div>
            </div>
          </div>

          <div className="overflow-x-auto max-h-[300px] overflow-y-auto border-t border-slate-200">
            <table className="w-full text-left border-collapse">
              <thead className="sticky top-0 bg-slate-50 border-b border-slate-200 text-[11px] font-medium text-slate-600">
                <tr>
                  <th className="py-2 px-2.5">PURE Card</th>
                  <th className="py-2 px-2.5 text-right">USD</th>
                  <th className="py-2 px-2.5 text-right">Base</th>
                  <th className="py-2 px-2.5 text-right">10x Return</th>
                  <th className="py-2 px-2.5 text-right">Action</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100 text-xs">
                {RECHARGE_PACKAGES.map((pkg) => (
                  <tr key={pkg.id} className="hover:bg-slate-50">
                    <td className="py-2.5 px-2.5 font-semibold text-slate-900 whitespace-nowrap">
                      {pkg.cardCode}
                    </td>
                    <td className="py-2.5 px-2.5 text-right font-mono tabular-nums text-slate-600">
                      ${pkg.usdPrice.toLocaleString()}
                    </td>
                    <td className="py-2.5 px-2.5 text-right font-mono tabular-nums text-slate-600">
                      {formatCoins(pkg.baseCoins)}
                    </td>
                    <td className="py-2.5 px-2.5 text-right font-mono tabular-nums font-semibold text-amber-700 whitespace-nowrap">
                      {formatMillionsLabel(pkg.totalCoins)}
                    </td>
                    <td className="py-2.5 px-2.5 text-right">
                      <button
                        type="button"
                        onClick={() =>
                          onRechargePackage(pkg.totalCoins, pkg.cardCode, pkg.usdPrice)
                        }
                        className="px-3 py-1.5 text-xs font-semibold text-blue-700 hover:text-white hover:bg-blue-600 border border-blue-200 rounded-md transition-colors whitespace-nowrap cursor-pointer"
                      >
                        Redeem 10x
                      </button>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </section>
      </div>

      {/* 100 Gift Tiers + Multi-Tier Lucky Backs up to 20M Coins */}
      <section className="bg-white border border-slate-200 rounded-xl p-6 space-y-5">
        <div className="flex flex-wrap items-center justify-between gap-4">
          <div>
            <h2 className="font-display text-lg font-semibold text-slate-900">
              100 Gift Tiers &amp; 20M Lucky Back Architecture
            </h2>
            <p className="text-xs text-slate-600 mt-0.5">
              Showing {filteredGifts.length} of {gifts.length} tiers · {economy.receiverCoinBackPct}% receiver coin-back · {economy.platformCommissionPct}% platform commission · {economy.luckyPoolPct}% lucky pool · {economy.seatSpSplitPct}% three-seat SP split.
            </p>
          </div>

          <div className="relative w-full sm:w-64">
            <Search className="w-3.5 h-3.5 text-slate-400 absolute left-3 top-1/2 -translate-y-1/2" />
            <input
              type="text"
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              placeholder="Search tier or gift name..."
              className="w-full pl-8 pr-3 py-2 text-xs bg-slate-50 border border-slate-200 rounded-lg focus:outline-none focus:border-blue-600"
            />
          </div>
        </div>

        {/* Bracket Filter Tabs */}
        <div className="flex flex-wrap items-center gap-1 p-1 bg-slate-100 rounded-lg w-fit">
          {(
            [
              ['all', 'All 100+'],
              ['1-25', 'Tiers 1–25'],
              ['26-50', 'Tiers 26–50'],
              ['51-75', 'Tiers 51–75'],
              ['76-100', 'Tiers 76–100'],
              ['rockets', 'Rockets'],
              ['luckybacks', 'Lucky Backs (to 20M)'],
              ['custom', 'Custom 1M+'],
            ] as [FilterBracket, string][]
          ).map(([key, label]) => (
            <button
              key={key}
              type="button"
              onClick={() => setBracket(key)}
              className={`px-3 py-1.5 text-xs font-medium rounded-md transition-colors whitespace-nowrap shrink-0 cursor-pointer ${
                bracket === key
                  ? 'bg-white text-slate-900 shadow-sm'
                  : 'text-slate-600 hover:text-slate-900'
              }`}
            >
              {label}
            </button>
          ))}
        </div>

        {/* High-Density Data Table */}
        <div className="overflow-x-auto max-h-[520px] overflow-y-auto border-t border-slate-200">
          <table className="w-full text-left border-collapse">
            <thead className="sticky top-0 bg-slate-50 border-b border-slate-200 text-[11px] font-medium text-slate-600">
              <tr>
                <th className="py-2.5 px-3">Tier</th>
                <th className="py-2.5 px-3">Gift Name</th>
                <th className="py-2.5 px-3">Type</th>
                <th className="py-2.5 px-3 text-right">Coin Cost</th>
                <th className="py-2.5 px-3 text-right">40% Coin-Back</th>
                <th className="py-2.5 px-3 text-right">15% Platform</th>
                <th className="py-2.5 px-3 text-right">5% Lucky Pool</th>
                <th className="py-2.5 px-3 text-right">2% 3-Seat SP</th>
                <th className="py-2.5 px-3 text-right">Catchers</th>
                <th className="py-2.5 px-3 text-right">Send</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-slate-100 text-xs">
              {filteredGifts.map((gift) => {
                const split = calculateGiftSplit(gift.coinCost, economy);
                return (
                  <tr key={gift.id} className="hover:bg-slate-50">
                    <td className="py-2.5 px-3 font-mono tabular-nums text-slate-500">
                      #{gift.tier}
                    </td>
                    <td className="py-2.5 px-3 font-medium text-slate-900">
                      <span className="inline-flex items-center gap-1.5">
                        {gift.isRocket && <Rocket className="w-3.5 h-3.5 text-amber-600" />}
                        {gift.category === 'Custom' && (
                          <Sparkles className="w-3.5 h-3.5 text-blue-600" />
                        )}
                        {gift.name}
                      </span>
                    </td>
                    <td className="py-2.5 px-3 text-slate-600">{gift.category}</td>
                    <td className="py-2.5 px-3 text-right font-mono tabular-nums font-semibold text-slate-900">
                      {gift.coinCost.toLocaleString()}
                    </td>
                    <td className="py-2.5 px-3 text-right font-mono tabular-nums text-emerald-700">
                      {split.receiverCoinBack.toLocaleString()}
                    </td>
                    <td className="py-2.5 px-3 text-right font-mono tabular-nums text-slate-600">
                      {split.platformCommission.toLocaleString()}
                    </td>
                    <td className="py-2.5 px-3 text-right font-mono tabular-nums text-amber-700">
                      {split.luckyPoolTotal.toLocaleString()}
                    </td>
                    <td className="py-2.5 px-3 text-right font-mono tabular-nums text-blue-700">
                      {split.seatSpTotal.toLocaleString()}
                    </td>
                    <td className="py-2.5 px-3 text-right font-mono tabular-nums text-slate-600">
                      {gift.defaultCatcherLimit}
                    </td>
                    <td className="py-2.5 px-3 text-right">
                      <button
                        type="button"
                        disabled={user.coinBalance < gift.coinCost}
                        onClick={() =>
                          onSendGiftToBoat(gift, gift.defaultCatcherLimit)
                        }
                        className={`px-2.5 py-1 text-xs font-medium rounded-md transition-colors whitespace-nowrap ${
                          user.coinBalance >= gift.coinCost
                            ? 'bg-slate-900 hover:bg-blue-600 text-white cursor-pointer'
                            : 'bg-slate-100 text-slate-400 cursor-not-allowed'
                        }`}
                      >
                        Launch
                      </button>
                    </td>
                  </tr>
                );
              })}
            </tbody>
          </table>
        </div>
      </section>
    </div>
  );
};
