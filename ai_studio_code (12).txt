import React, { useState } from 'react';
import {
  CheckCircle2,
  Eye,
  EyeOff,
  Lock,
  Mic,
  MicOff,
  RotateCcw,
  ShieldAlert,
  Unlock,
  UserX,
} from 'lucide-react';
import { DEFAULT_ECONOMY_CONFIG } from '../data/economySpecs';
import {
  AuditLogEntry,
  BoatRoomState,
  EconomyConfig,
  MerchantTransaction,
  RoleType,
  SupportTicket,
  UserProfile,
} from '../types/architecture';
import { formatUidDisplay } from '../utils/economyMath';

interface ControlPanelsViewProps {
  activeRole: RoleType;
  onSelectRole: (role: RoleType) => void;
  economy: EconomyConfig;
  onUpdateEconomy: (next: EconomyConfig) => void;
  user: UserProfile;
  onUpdateUser: (patch: Partial<UserProfile>) => void;
  room: BoatRoomState;
  onUpdateRoom: (patch: Partial<BoatRoomState>) => void;
  onToggleSeatMute: (seatNum: 1 | 2 | 3) => void;
  onVacateSeat: (seatNum: 1 | 2 | 3) => void;
  tickets: SupportTicket[];
  onResolveTicket: (ticketId: string) => void;
  merchantTxs: MerchantTransaction[];
  onIssueMerchantTopUp: (targetUid: string, baseCoins: number) => void;
  auditLogs: AuditLogEntry[];
  onAppendAudit: (role: RoleType, action: string, target: string) => void;
}

const ROLES: { id: RoleType; label: string; scope: string }[] = [
  { id: 'Master', label: 'Master', scope: 'Platform Economy & Root Governance' },
  { id: 'SuperAdmin', label: 'Super Admin', scope: 'VIP/SVP Grants & Agency Overrides' },
  { id: 'Admin', label: 'Admin', scope: 'Live Boat Room Moderation' },
  { id: 'BoatOwner', label: 'Boat Owner', scope: 'Lock/Unlock, Anytime Hide ID & 3 Seats' },
  { id: 'Merchant', label: 'Merchant', scope: '10x Recharge Stock & UID Credits' },
  { id: 'CSSupport', label: 'CS Support', scope: 'Tickets, Claim Audits & UID Lookup' },
];

export const ControlPanelsView: React.FC<ControlPanelsViewProps> = ({
  activeRole,
  onSelectRole,
  economy,
  onUpdateEconomy,
  user,
  onUpdateUser,
  room,
  onUpdateRoom,
  onToggleSeatMute,
  onVacateSeat,
  tickets,
  onResolveTicket,
  merchantTxs,
  onIssueMerchantTopUp,
  auditLogs,
  onAppendAudit,
}) => {
  const [merchantTargetUid, setMerchantTargetUid] = useState(user.uid);
  const [merchantBaseCoins, setMerchantBaseCoins] = useState(100000);
  const [merchantInventory, setMerchantInventory] = useState(250_000_000);
  const [pinDraft, setPinDraft] = useState(room.lockPin);
  const [lookupUid, setLookupUid] = useState(user.uid);
  const [merchantAgencyApproved, setMerchantAgencyApproved] = useState(true);

  const currentRoleMeta = ROLES.find((r) => r.id === activeRole)!;

  const handleMerchantSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    const totalCredited = merchantBaseCoins * economy.rechargeBonusMultiplier;
    if (totalCredited > merchantInventory || merchantBaseCoins <= 0) return;
    setMerchantInventory((prev) => prev - totalCredited);
    onIssueMerchantTopUp(merchantTargetUid.trim() || user.uid, merchantBaseCoins);
  };

  return (
    <div className="space-y-6">
      {/* 6-Role Switcher Bar */}
      <section className="bg-white border border-slate-200 rounded-xl p-6 space-y-4">
        <div className="flex flex-wrap items-center justify-between gap-4">
          <div>
            <h2 className="font-display text-lg font-semibold text-slate-900">
              RBAC Operations Console — {currentRoleMeta.label}
            </h2>
            <p className="text-xs text-slate-600 mt-0.5">{currentRoleMeta.scope}</p>
          </div>
          <div className="text-xs font-mono tabular-nums text-slate-500">
            6 Role Scopes Active
          </div>
        </div>

        <div className="flex flex-wrap items-center gap-1 p-1 bg-slate-100 rounded-lg w-fit">
          {ROLES.map((role) => (
            <button
              key={role.id}
              type="button"
              onClick={() => onSelectRole(role.id)}
              className={`px-3.5 py-2 text-xs font-medium rounded-md transition-colors whitespace-nowrap shrink-0 cursor-pointer ${
                activeRole === role.id
                  ? 'bg-white text-slate-900 shadow-sm'
                  : 'text-slate-600 hover:text-slate-900'
              }`}
            >
              {role.label}
            </button>
          ))}
        </div>
      </section>

      {/* Role Workspace */}
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-6 items-start">
        <div className="lg:col-span-8 bg-white border border-slate-200 rounded-xl p-6 space-y-6">
          {/* 1. MASTER CONTROL PANEL */}
          {activeRole === 'Master' && (
            <div className="space-y-5">
              <div className="flex items-center justify-between gap-4">
                <div>
                  <h3 className="font-display text-base font-semibold text-slate-900">
                    Global Economy &amp; Split Parameters
                  </h3>
                  <p className="text-xs text-slate-600 mt-0.5">
                    Adjust core percentages live. Changes immediately affect all Boat Room gift launches.
                  </p>
                </div>
                <button
                  type="button"
                  onClick={() => {
                    onUpdateEconomy(DEFAULT_ECONOMY_CONFIG);
                    onAppendAudit(
                      'Master',
                      'Reset economy splits to specification defaults (40/15/5/2)',
                      'Global Engine'
                    );
                  }}
                  className="inline-flex items-center gap-1.5 px-3 py-1.5 text-xs font-medium border border-slate-200 hover:bg-slate-50 rounded-lg cursor-pointer"
                >
                  <RotateCcw className="w-3.5 h-3.5" />
                  Reset Defaults
                </button>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 gap-4">
                <label className="block text-xs">
                  <span className="text-slate-600 font-medium">
                    Receiver Coin-Back (%)
                  </span>
                  <input
                    type="number"
                    min={0}
                    max={70}
                    value={economy.receiverCoinBackPct}
                    onChange={(e) =>
                      onUpdateEconomy({
                        ...economy,
                        receiverCoinBackPct: Number(e.target.value),
                      })
                    }
                    className="mt-1 w-full px-3 py-2 font-mono tabular-nums bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </label>

                <label className="block text-xs">
                  <span className="text-slate-600 font-medium">
                    Platform Commission (%)
                  </span>
                  <input
                    type="number"
                    min={0}
                    max={50}
                    value={economy.platformCommissionPct}
                    onChange={(e) =>
                      onUpdateEconomy({
                        ...economy,
                        platformCommissionPct: Number(e.target.value),
                      })
                    }
                    className="mt-1 w-full px-3 py-2 font-mono tabular-nums bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </label>

                <label className="block text-xs">
                  <span className="text-slate-600 font-medium">
                    Dynamic Lucky Pool (%)
                  </span>
                  <input
                    type="number"
                    min={1}
                    max={25}
                    value={economy.luckyPoolPct}
                    onChange={(e) =>
                      onUpdateEconomy({
                        ...economy,
                        luckyPoolPct: Number(e.target.value),
                      })
                    }
                    className="mt-1 w-full px-3 py-2 font-mono tabular-nums bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </label>

                <label className="block text-xs">
                  <span className="text-slate-600 font-medium">
                    3-Seat SP Split (%)
                  </span>
                  <input
                    type="number"
                    min={1}
                    max={15}
                    value={economy.seatSpSplitPct}
                    onChange={(e) =>
                      onUpdateEconomy({
                        ...economy,
                        seatSpSplitPct: Number(e.target.value),
                      })
                    }
                    className="mt-1 w-full px-3 py-2 font-mono tabular-nums bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </label>

                <label className="block text-xs">
                  <span className="text-slate-600 font-medium">
                    Min Catcher Limit
                  </span>
                  <input
                    type="number"
                    min={5}
                    max={25}
                    value={economy.minCatcherLimit}
                    onChange={(e) =>
                      onUpdateEconomy({
                        ...economy,
                        minCatcherLimit: Number(e.target.value),
                      })
                    }
                    className="mt-1 w-full px-3 py-2 font-mono tabular-nums bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </label>

                <label className="block text-xs">
                  <span className="text-slate-600 font-medium">
                    Max Catcher Limit
                  </span>
                  <input
                    type="number"
                    min={20}
                    max={100}
                    value={economy.maxCatcherLimit}
                    onChange={(e) =>
                      onUpdateEconomy({
                        ...economy,
                        maxCatcherLimit: Number(e.target.value),
                      })
                    }
                    className="mt-1 w-full px-3 py-2 font-mono tabular-nums bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </label>
              </div>
            </div>
          )}

          {/* 2. SUPER ADMIN CONTROL PANEL */}
          {activeRole === 'SuperAdmin' && (
            <div className="space-y-5">
              <div>
                <h3 className="font-display text-base font-semibold text-slate-900">
                  Privilege Grants &amp; Agency Oversight
                </h3>
                <p className="text-xs text-slate-600 mt-0.5">
                  Directly adjust target user Level (1–200), VIP (1–5), SVIP (1–15), and Merchant authorization.
                </p>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
                <label className="block text-xs">
                  <span className="text-slate-600 font-medium">
                    User Level (1–200)
                  </span>
                  <input
                    type="number"
                    min={1}
                    max={200}
                    value={user.level}
                    onChange={(e) => {
                      const val = Math.min(200, Math.max(1, Number(e.target.value)));
                      onUpdateUser({ level: val });
                      onAppendAudit('SuperAdmin', `Set Level to ${val}`, `UID ${user.uid}`);
                    }}
                    className="mt-1 w-full px-3 py-2 font-mono tabular-nums bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </label>

                <label className="block text-xs">
                  <span className="text-slate-600 font-medium">
                    VIP Tier (0–5) · Hide ID at 5
                  </span>
                  <input
                    type="number"
                    min={0}
                    max={5}
                    value={user.vipLevel}
                    onChange={(e) => {
                      const val = Math.min(5, Math.max(0, Number(e.target.value)));
                      onUpdateUser({ vipLevel: val });
                      onAppendAudit('SuperAdmin', `Granted VIP ${val}`, `UID ${user.uid}`);
                    }}
                    className="mt-1 w-full px-3 py-2 font-mono tabular-nums bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </label>

                <label className="block text-xs">
                  <span className="text-slate-600 font-medium">
                    SVIP Tier (0–15) · 1M Gift at 4+
                  </span>
                  <input
                    type="number"
                    min={0}
                    max={15}
                    value={user.svpLevel}
                    onChange={(e) => {
                      const val = Math.min(15, Math.max(0, Number(e.target.value)));
                      onUpdateUser({ svpLevel: val });
                      onAppendAudit('SuperAdmin', `Granted SVIP ${val}`, `UID ${user.uid}`);
                    }}
                    className="mt-1 w-full px-3 py-2 font-mono tabular-nums bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </label>
              </div>

              <div className="pt-4 border-t border-slate-100 flex flex-wrap items-center justify-between gap-4">
                <div>
                  <div className="text-xs font-semibold text-slate-900">
                    Merchant Desk Authorization · Pacific Coin Desk
                  </div>
                  <div className="text-xs text-slate-500 mt-0.5">
                    Status: {merchantAgencyApproved ? 'Authorized for 10x Recharge Issuance' : 'Suspended'}
                  </div>
                </div>
                <button
                  type="button"
                  onClick={() => {
                    const next = !merchantAgencyApproved;
                    setMerchantAgencyApproved(next);
                    onAppendAudit(
                      'SuperAdmin',
                      next ? 'Approved Merchant Desk' : 'Suspended Merchant Desk',
                      'Pacific Coin Desk'
                    );
                  }}
                  className={`px-4 py-2 text-xs font-semibold rounded-lg transition-colors cursor-pointer ${
                    merchantAgencyApproved
                      ? 'bg-red-50 text-red-700 border border-red-200 hover:bg-red-100'
                      : 'bg-emerald-600 text-white hover:bg-emerald-700'
                  }`}
                >
                  {merchantAgencyApproved ? 'Suspend Merchant' : 'Approve Merchant'}
                </button>
              </div>
            </div>
          )}

          {/* 3. ADMIN MODERATION PANEL */}
          {activeRole === 'Admin' && (
            <div className="space-y-5">
              <div>
                <h3 className="font-display text-base font-semibold text-slate-900">
                  Live Boat Room Moderation ({room.roomId})
                </h3>
                <p className="text-xs text-slate-600 mt-0.5">
                  Override room lock states, mute stage seats, or dispatch compliance notices.
                </p>
              </div>

              <div className="flex flex-wrap items-center gap-3">
                <button
                  type="button"
                  onClick={() => {
                    onUpdateRoom({ isLocked: !room.isLocked });
                    onAppendAudit(
                      'Admin',
                      room.isLocked ? 'Force Unlocked Boat Room' : 'Force Locked Boat Room',
                      room.roomId
                    );
                  }}
                  className="inline-flex items-center gap-1.5 px-4 py-2 text-xs font-semibold bg-slate-900 hover:bg-slate-800 text-white rounded-lg cursor-pointer"
                >
                  {room.isLocked ? <Unlock className="w-3.5 h-3.5" /> : <Lock className="w-3.5 h-3.5" />}
                  {room.isLocked ? 'Force Unlock Boat' : 'Force Lock Boat'}
                </button>

                <button
                  type="button"
                  onClick={() => {
                    onUpdateRoom({
                      chatMessages: [
                        ...room.chatMessages,
                        {
                          id: `mod-${Date.now()}`,
                          senderName: 'Room Admin',
                          senderUid: '000000',
                          isMasked: false,
                          text: 'Moderation notice: Keep lucky pool claims and seat audio compliant.',
                          timestamp: new Date().toLocaleTimeString('en-US', {
                            hour12: false,
                            hour: '2-digit',
                            minute: '2-digit',
                          }),
                          type: 'system',
                        },
                      ],
                    });
                    onAppendAudit('Admin', 'Broadcast moderation notice', room.roomId);
                  }}
                  className="inline-flex items-center gap-1.5 px-4 py-2 text-xs font-medium border border-slate-200 hover:bg-slate-50 rounded-lg cursor-pointer"
                >
                  <ShieldAlert className="w-3.5 h-3.5 text-amber-600" />
                  Send Compliance Notice
                </button>
              </div>

              <div className="divide-y divide-slate-100 border-t border-slate-200">
                {room.seats.map((seat) => (
                  <div key={seat.seatNumber} className="py-3 flex items-center justify-between gap-3 text-xs">
                    <div>
                      <span className="font-semibold text-slate-900">
                        Seat {seat.seatNumber}
                      </span>{' '}
                      <span className="text-slate-600">
                        · {seat.occupantName || 'Empty'} {seat.occupantUid ? `(UID ${seat.occupantUid})` : ''}
                      </span>
                    </div>
                    <div className="flex items-center gap-2">
                      <button
                        type="button"
                        onClick={() => onToggleSeatMute(seat.seatNumber)}
                        className="px-3 py-1.5 border border-slate-200 hover:bg-slate-50 rounded-md inline-flex items-center gap-1 cursor-pointer"
                      >
                        {seat.isMuted ? <MicOff className="w-3.5 h-3.5 text-red-600" /> : <Mic className="w-3.5 h-3.5 text-emerald-600" />}
                        {seat.isMuted ? 'Unmute' : 'Mute'}
                      </button>
                      {seat.occupantName && (
                        <button
                          type="button"
                          onClick={() => onVacateSeat(seat.seatNumber)}
                          className="px-3 py-1.5 border border-red-200 text-red-700 hover:bg-red-50 rounded-md inline-flex items-center gap-1 cursor-pointer"
                        >
                          <UserX className="w-3.5 h-3.5" />
                          Remove
                        </button>
                      )}
                    </div>
                  </div>
                ))}
              </div>
            </div>
          )}

          {/* 4. BOAT OWNER CONTROL PANEL */}
          {activeRole === 'BoatOwner' && (
            <div className="space-y-5">
              <div className="flex flex-wrap items-center justify-between gap-4">
                <div>
                  <h3 className="font-display text-base font-semibold text-slate-900">
                    Boat Owner Deck Controls ({room.roomName})
                  </h3>
                  <p className="text-xs text-slate-600 mt-0.5">
                    Owner anytime Hide ID override, Boat Lock/Unlock PIN, and 40% Receiver Coin-Back vault.
                  </p>
                </div>
                <div className="text-right font-mono tabular-nums">
                  <div className="text-[11px] text-slate-500">40% Coin-Back Earned</div>
                  <div className="text-sm font-semibold text-emerald-700">
                    {room.totalReceiverCoinBack.toLocaleString()} coins
                  </div>
                </div>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                <div className="p-4 border border-slate-200 rounded-lg space-y-3">
                  <div className="text-xs font-semibold text-slate-900">
                    Owner Anytime Hide ID
                  </div>
                  <p className="text-xs text-slate-600">
                    Boat Owners can mask their UID ({formatUidDisplay(room.hostUid, room.ownerHideId)}) at any time regardless of VIP level.
                  </p>
                  <button
                    type="button"
                    onClick={() => {
                      onUpdateRoom({ ownerHideId: !room.ownerHideId });
                      onAppendAudit(
                        'BoatOwner',
                        room.ownerHideId ? 'Disabled Owner Hide ID' : 'Enabled Owner Hide ID',
                        room.roomId
                      );
                    }}
                    className="inline-flex items-center gap-1.5 px-4 py-2 text-xs font-semibold bg-slate-900 hover:bg-slate-800 text-white rounded-lg cursor-pointer"
                  >
                    {room.ownerHideId ? <EyeOff className="w-3.5 h-3.5" /> : <Eye className="w-3.5 h-3.5" />}
                    {room.ownerHideId ? 'Show Owner ID' : 'Hide Owner ID Now'}
                  </button>
                </div>

                <div className="p-4 border border-slate-200 rounded-lg space-y-3">
                  <div className="text-xs font-semibold text-slate-900">
                    Boat Lock / Unlock System
                  </div>
                  <div className="flex items-center gap-2">
                    <input
                      type="text"
                      maxLength={6}
                      value={pinDraft}
                      onChange={(e) => setPinDraft(e.target.value)}
                      className="w-24 px-2.5 py-1.5 text-xs font-mono border border-slate-200 rounded-md"
                    />
                    <button
                      type="button"
                      onClick={() => {
                        onUpdateRoom({
                          lockPin: pinDraft || '4820',
                          isLocked: !room.isLocked,
                        });
                        onAppendAudit(
                          'BoatOwner',
                          room.isLocked ? 'Unlocked Boat Room' : `Locked Boat Room (PIN ${pinDraft})`,
                          room.roomId
                        );
                      }}
                      className={`inline-flex items-center gap-1.5 px-4 py-1.5 text-xs font-semibold rounded-md cursor-pointer ${
                        room.isLocked
                          ? 'bg-emerald-600 hover:bg-emerald-700 text-white'
                          : 'bg-red-600 hover:bg-red-700 text-white'
                      }`}
                    >
                      {room.isLocked ? <Unlock className="w-3.5 h-3.5" /> : <Lock className="w-3.5 h-3.5" />}
                      {room.isLocked ? 'Unlock Boat' : 'Lock Boat'}
                    </button>
                  </div>
                </div>
              </div>
            </div>
          )}

          {/* 5. MERCHANT CONTROL PANEL */}
          {activeRole === 'Merchant' && (
            <div className="space-y-5">
              <div className="flex flex-wrap items-center justify-between gap-4">
                <div>
                  <h3 className="font-display text-base font-semibold text-slate-900">
                    Merchant 10x Coin Issuance Desk
                  </h3>
                  <p className="text-xs text-slate-600 mt-0.5">
                    Issue 10x recharge bonus packages directly to user UIDs from merchant inventory.
                  </p>
                </div>
                <div className="text-right font-mono tabular-nums">
                  <div className="text-[11px] text-slate-500">Merchant Vault Stock</div>
                  <div className="text-sm font-semibold text-slate-900">
                    {merchantInventory.toLocaleString()} coins
                  </div>
                </div>
              </div>

              <form onSubmit={handleMerchantSubmit} className="grid grid-cols-1 sm:grid-cols-3 gap-3 items-end">
                <label className="block text-xs">
                  <span className="text-slate-600 font-medium">Target User UID</span>
                  <input
                    type="text"
                    value={merchantTargetUid}
                    onChange={(e) => setMerchantTargetUid(e.target.value)}
                    className="mt-1 w-full px-3 py-2 font-mono bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </label>

                <label className="block text-xs">
                  <span className="text-slate-600 font-medium">
                    Base Coins (10x = {(merchantBaseCoins * economy.rechargeBonusMultiplier).toLocaleString()})
                  </span>
                  <input
                    type="number"
                    min={1000}
                    step={5000}
                    value={merchantBaseCoins}
                    onChange={(e) => setMerchantBaseCoins(Number(e.target.value))}
                    className="mt-1 w-full px-3 py-2 font-mono tabular-nums bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </label>

                <button
                  type="submit"
                  className="px-4 py-2 text-xs font-semibold bg-blue-600 hover:bg-blue-700 text-white rounded-lg transition-colors cursor-pointer"
                >
                  Transfer 10x Package
                </button>
              </form>

              <div className="overflow-x-auto border-t border-slate-200 pt-3">
                <table className="w-full text-left border-collapse">
                  <thead>
                    <tr className="border-b border-slate-200 text-[11px] font-medium text-slate-500">
                      <th className="py-2 pr-3">Tx ID</th>
                      <th className="py-2 px-3">Target UID</th>
                      <th className="py-2 px-3 text-right">Base</th>
                      <th className="py-2 px-3 text-right">Bonus</th>
                      <th className="py-2 pl-3 text-right">Total 10x</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-slate-100 text-xs">
                    {merchantTxs.map((tx) => (
                      <tr key={tx.id}>
                        <td className="py-2 pr-3 font-mono text-slate-500">{tx.id}</td>
                        <td className="py-2 px-3 font-mono text-slate-900">{tx.targetUid}</td>
                        <td className="py-2 px-3 text-right font-mono tabular-nums text-slate-600">
                          {tx.baseCoins.toLocaleString()}
                        </td>
                        <td className="py-2 px-3 text-right font-mono tabular-nums text-amber-700">
                          +{tx.bonusCoins.toLocaleString()}
                        </td>
                        <td className="py-2 pl-3 text-right font-mono tabular-nums font-semibold text-emerald-700">
                          {tx.totalCredited.toLocaleString()}
                        </td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
            </div>
          )}

          {/* 6. CS SUPPORT CONTROL PANEL */}
          {activeRole === 'CSSupport' && (
            <div className="space-y-5">
              <div className="flex flex-wrap items-center justify-between gap-4">
                <div>
                  <h3 className="font-display text-base font-semibold text-slate-900">
                    CS Support Ticket &amp; UID Verification Queue
                  </h3>
                  <p className="text-xs text-slate-600 mt-0.5">
                    Investigate 10x recharge credits, Lucky Pool payouts, VIP 5 Hide ID eligibility, and Follow Video Call permissions.
                  </p>
                </div>

                <div className="flex items-center gap-2">
                  <input
                    type="text"
                    value={lookupUid}
                    onChange={(e) => setLookupUid(e.target.value)}
                    placeholder="Lookup UID..."
                    className="w-32 px-3 py-1.5 text-xs font-mono bg-slate-50 border border-slate-200 rounded-lg"
                  />
                </div>
              </div>

              {lookupUid === user.uid && (
                <div className="p-3 bg-slate-50 border border-slate-200 rounded-lg text-xs font-mono tabular-nums text-slate-700">
                  UID {user.uid} ({user.name}) · Lv.{user.level} · VIP {user.vipLevel} (Hide ID {user.vipLevel >= 5 ? 'Eligible' : 'Locked'}) · SVP {user.svpLevel} (1M Gift {user.svpLevel >= 4 || user.level >= 150 ? 'Unlocked' : 'Locked'}) · Balance {user.coinBalance.toLocaleString()}
                </div>
              )}

              <div className="divide-y divide-slate-100 border-t border-slate-200">
                {tickets.map((ticket) => (
                  <div key={ticket.id} className="py-3.5 flex flex-wrap items-center justify-between gap-4">
                    <div className="space-y-1 max-w-xl">
                      <div className="flex items-center gap-2 text-xs">
                        <span className="font-mono font-semibold text-slate-900">
                          {ticket.id}
                        </span>
                        <span aria-hidden="true">·</span>
                        <span className="text-slate-600">{ticket.category}</span>
                        <span aria-hidden="true">·</span>
                        <span className="font-mono text-slate-500">
                          UID {ticket.userUid} ({ticket.userName})
                        </span>
                      </div>
                      <p className="text-xs text-slate-700">{ticket.summary}</p>
                    </div>

                    <div className="flex items-center gap-3">
                      <span
                        className={`text-xs font-medium ${
                          ticket.status === 'Resolved'
                            ? 'text-emerald-700'
                            : ticket.status === 'Investigating'
                              ? 'text-amber-700'
                              : 'text-blue-700'
                        }`}
                      >
                        {ticket.status}
                      </span>
                      {ticket.status !== 'Resolved' && (
                        <button
                          type="button"
                          onClick={() => onResolveTicket(ticket.id)}
                          className="inline-flex items-center gap-1 px-3 py-1.5 text-xs font-semibold bg-emerald-600 hover:bg-emerald-700 text-white rounded-md cursor-pointer"
                        >
                          <CheckCircle2 className="w-3.5 h-3.5" />
                          Resolve
                        </button>
                      )}
                    </div>
                  </div>
                ))}
              </div>
            </div>
          )}
        </div>

        {/* Right Column: Live Audit Log */}
        <aside className="lg:col-span-4 bg-white border border-slate-200 rounded-xl p-5 space-y-4">
          <div>
            <h3 className="font-display text-sm font-semibold text-slate-900">
              RBAC Action Audit Trail
            </h3>
            <p className="text-xs text-slate-500 mt-0.5">
              Immutable ledger of control panel mutations across all 6 roles.
            </p>
          </div>

          <div className="divide-y divide-slate-100 max-h-[380px] overflow-y-auto">
            {auditLogs.map((log) => (
              <div key={log.id} className="py-2.5 text-xs">
                <div className="flex items-center justify-between text-slate-500 font-mono tabular-nums">
                  <span className="font-semibold text-slate-800">{log.actorRole}</span>
                  <span>{log.timestamp}</span>
                </div>
                <div className="text-slate-700 mt-0.5">{log.action}</div>
                <div className="text-[11px] font-mono text-slate-500 mt-0.5">
                  Target: {log.target}
                </div>
              </div>
            ))}
          </div>
        </aside>
      </div>
    </div>
  );
};
