import React, { useMemo, useState } from 'react';
import {
  Eye,
  EyeOff,
  Lock,
  Mic,
  MicOff,
  PhoneCall,
  PhoneOff,
  Rocket,
  Send,
  Unlock,
  UserCheck,
  UserPlus,
  Video,
  VideoOff,
} from 'lucide-react';
import { ASSET_PATHS } from '../data/economySpecs';
import {
  BoatRoomState,
  EconomyConfig,
  GiftTier,
  RocketLaunchEvent,
  UserProfile,
} from '../types/architecture';
import {
  calculateGiftSplit,
  canHideId,
  formatCoins,
  formatUidDisplay,
} from '../utils/economyMath';

interface BoatRoomViewProps {
  user: UserProfile;
  room: BoatRoomState;
  economy: EconomyConfig;
  gifts: GiftTier[];
  activeRocket: RocketLaunchEvent | null;
  launchHistory: RocketLaunchEvent[];
  onToggleFollow: () => void;
  onToggleBoatLock: (pinInput?: string) => void;
  onToggleOwnerHideId: () => void;
  onToggleUserHideId: () => void;
  onLaunchGift: (gift: GiftTier, catcherLimit: number) => void;
  onClaimLuckyDrop: () => void;
  onSimulateRemainingClaims: () => void;
  onSendChat: (message: string) => void;
  onToggleSeatMute: (seatNum: 1 | 2 | 3) => void;
}

export const BoatRoomView: React.FC<BoatRoomViewProps> = ({
  user,
  room,
  economy,
  gifts,
  activeRocket,
  launchHistory,
  onToggleFollow,
  onToggleBoatLock,
  onToggleOwnerHideId,
  onToggleUserHideId,
  onLaunchGift,
  onClaimLuckyDrop,
  onSimulateRemainingClaims,
  onSendChat,
  onToggleSeatMute,
}) => {
  const featuredRockets = useMemo(() => {
    const custom = gifts.filter((g) => g.category === 'Custom');
    const rocketsAndLucky = gifts.filter(
      (g) =>
        g.tier === 50 ||
        g.tier === 75 ||
        g.tier === 85 ||
        g.tier === 95 ||
        g.tier === 100
    );
    return [...custom, ...rocketsAndLucky];
  }, [gifts]);

  const [selectedGiftId, setSelectedGiftId] = useState<string>(
    featuredRockets[featuredRockets.length - 1]?.id || gifts[99].id
  );
  const selectedGift =
    gifts.find((g) => g.id === selectedGiftId) || featuredRockets[0] || gifts[99];

  const [catcherLimit, setCatcherLimit] = useState<number>(
    selectedGift.defaultCatcherLimit || 25
  );
  const [chatInput, setChatInput] = useState('');
  const [videoCallOpen, setVideoCallOpen] = useState(false);
  const [callMuted, setCallMuted] = useState(false);
  const [callCameraOff, setCallCameraOff] = useState(false);
  const [pinEntry, setPinEntry] = useState('');
  const [roomAccessGranted, setRoomAccessGranted] = useState(true);

  const previewSplit = useMemo(
    () => calculateGiftSplit(selectedGift.coinCost, economy),
    [selectedGift.coinCost, economy]
  );

  const userCanHide = canHideId(user, false);
  const hasUserClaimedActive = activeRocket
    ? activeRocket.claims.some((c) => c.claimerUid === user.uid)
    : false;

  const handleGiftSelect = (gift: GiftTier) => {
    setSelectedGiftId(gift.id);
    setCatcherLimit(
      Math.min(
        economy.maxCatcherLimit,
        Math.max(economy.minCatcherLimit, gift.defaultCatcherLimit)
      )
    );
  };

  const handleSendChatSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    const trimmed = chatInput.trim();
    if (!trimmed) return;
    onSendChat(trimmed);
    setChatInput('');
  };

  const handleUnlockWithPin = (e: React.FormEvent) => {
    e.preventDefault();
    if (pinEntry === room.lockPin) {
      setRoomAccessGranted(true);
      setPinEntry('');
    }
  };

  const handleToggleLock = () => {
    const nextLocked = !room.isLocked;
    onToggleBoatLock();
    if (nextLocked) {
      setRoomAccessGranted(false);
    } else {
      setRoomAccessGranted(true);
    }
  };

  return (
    <div className="space-y-6">
      {/* Main 2-column Broadcast & Feed Grid */}
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-6 items-stretch">
        {/* Left/Center Live Boat Stage */}
        <div className="lg:col-span-8 flex flex-col gap-6">
          <div className="relative rounded-xl overflow-hidden border border-slate-200 bg-slate-950 text-white">
            {/* Backdrop image with resilient fallback & scrim */}
            <div className="relative min-h-[430px] flex flex-col justify-between p-6">
              <img
                src={ASSET_PATHS.boatBackdrop}
                alt="Harbor Sovereign Deck studio backdrop"
                referrerPolicy="no-referrer"
                className="absolute inset-0 w-full h-full object-cover opacity-55"
              />
              <div
                aria-hidden="true"
                className="absolute inset-0 bg-gradient-to-t from-slate-950 via-slate-950/65 to-slate-950/45"
              />

              {/* Top Stage Bar: Host Info + Follow/Video Call + Lock/Hide Controls */}
              <div className="relative z-10 flex flex-wrap items-center justify-between gap-4">
                <div className="flex items-center gap-3">
                  <div className="w-11 h-11 rounded-full overflow-hidden border border-white/25 bg-slate-800 shrink-0">
                    <img
                      src={room.hostAvatarUrl}
                      alt={room.hostName}
                      referrerPolicy="no-referrer"
                      className="w-full h-full object-cover"
                    />
                  </div>
                  <div>
                    <div className="flex items-center gap-2">
                      <h2 className="font-display text-base font-semibold tracking-tight text-white">
                        {room.roomName}
                      </h2>
                    </div>
                    <div className="flex flex-wrap items-center gap-2 text-xs text-slate-300 font-mono tabular-nums">
                      <span>{room.hostName}</span>
                      <span aria-hidden="true">·</span>
                      <span>{formatUidDisplay(room.hostUid, room.ownerHideId)}</span>
                      <span aria-hidden="true">·</span>
                      <span>Lv.{room.hostLevel}</span>
                      <span aria-hidden="true">·</span>
                      <span>SVP {room.hostSvp}</span>
                      <span aria-hidden="true">·</span>
                      <span>{room.viewerCount.toLocaleString()} watching</span>
                    </div>
                  </div>
                </div>

                <div className="flex flex-wrap items-center gap-2">
                  <button
                    type="button"
                    onClick={onToggleFollow}
                    className={`inline-flex items-center gap-1.5 px-3.5 py-2 text-xs font-semibold rounded-lg transition-colors whitespace-nowrap shrink-0 cursor-pointer ${
                      user.isFollowingHost
                        ? 'bg-white/15 text-white hover:bg-white/25 border border-white/20'
                        : 'bg-blue-600 text-white hover:bg-blue-500'
                    }`}
                  >
                    {user.isFollowingHost ? (
                      <>
                        <UserCheck className="w-3.5 h-3.5" />
                        Following
                      </>
                    ) : (
                      <>
                        <UserPlus className="w-3.5 h-3.5" />
                        Follow Host
                      </>
                    )}
                  </button>

                  <button
                    type="button"
                    disabled={!user.isFollowingHost}
                    onClick={() => setVideoCallOpen(true)}
                    title={
                      user.isFollowingHost
                        ? 'Start 1-on-1 Video Call with host'
                        : 'Follow host to unlock Video Call option'
                    }
                    className={`inline-flex items-center gap-1.5 px-3.5 py-2 text-xs font-semibold rounded-lg transition-colors whitespace-nowrap shrink-0 ${
                      user.isFollowingHost
                        ? 'bg-emerald-600 hover:bg-emerald-500 text-white cursor-pointer'
                        : 'bg-slate-800/80 text-slate-400 border border-white/10 cursor-not-allowed'
                    }`}
                  >
                    <Video className="w-3.5 h-3.5" />
                    {user.isFollowingHost ? 'Video Call' : 'Follow for Video'}
                  </button>

                  <button
                    type="button"
                    onClick={onToggleOwnerHideId}
                    className="inline-flex items-center gap-1.5 px-3 py-2 text-xs font-medium rounded-lg bg-slate-900/80 hover:bg-slate-800 text-slate-200 border border-white/15 transition-colors whitespace-nowrap shrink-0 cursor-pointer"
                  >
                    {room.ownerHideId ? (
                      <>
                        <EyeOff className="w-3.5 h-3.5 text-amber-400" />
                        Owner ID Hidden
                      </>
                    ) : (
                      <>
                        <Eye className="w-3.5 h-3.5" />
                        Owner Hide ID
                      </>
                    )}
                  </button>

                  <button
                    type="button"
                    onClick={handleToggleLock}
                    className={`inline-flex items-center gap-1.5 px-3 py-2 text-xs font-medium rounded-lg border transition-colors whitespace-nowrap shrink-0 cursor-pointer ${
                      room.isLocked
                        ? 'bg-red-950/80 border-red-500/40 text-red-200 hover:bg-red-900/80'
                        : 'bg-slate-900/80 border-white/15 text-slate-200 hover:bg-slate-800'
                    }`}
                  >
                    {room.isLocked ? (
                      <>
                        <Lock className="w-3.5 h-3.5" />
                        Locked (PIN {room.lockPin})
                      </>
                    ) : (
                      <>
                        <Unlock className="w-3.5 h-3.5" />
                        Boat Open
                      </>
                    )}
                  </button>
                </div>
              </div>

              {/* Center Stage: Locked State Gate OR Active Rocket Trajectory + Dynamic 5% Lucky Pool */}
              <div className="relative z-10 my-6">
                {room.isLocked && !roomAccessGranted ? (
                  <div className="max-w-md mx-auto bg-slate-900/90 border border-white/15 rounded-xl p-5 text-center space-y-3">
                    <Lock className="w-6 h-6 text-amber-400 mx-auto" />
                    <h3 className="font-display text-base font-semibold">
                      Boat Room Locked by Owner
                    </h3>
                    <p className="text-xs text-slate-300">
                      Enter room PIN ({room.lockPin}) or unlock from Boat Owner controls to resume stage interaction.
                    </p>
                    <form onSubmit={handleUnlockWithPin} className="flex items-center gap-2 justify-center">
                      <input
                        type="text"
                        value={pinEntry}
                        onChange={(e) => setPinEntry(e.target.value)}
                        placeholder="Enter PIN (4820)"
                        className="w-36 px-3 py-2 text-xs font-mono bg-slate-950 border border-white/20 rounded-lg text-white placeholder:text-slate-500 focus:outline-none focus:border-blue-500"
                      />
                      <button
                        type="submit"
                        className="px-4 py-2 text-xs font-semibold bg-blue-600 hover:bg-blue-500 rounded-lg transition-colors cursor-pointer"
                      >
                        Enter Boat
                      </button>
                    </form>
                  </div>
                ) : activeRocket ? (
                  <div className="bg-slate-900/85 border border-amber-400/30 rounded-xl p-5 space-y-4">
                    <div className="flex flex-wrap items-center justify-between gap-3">
                      <div className="flex items-center gap-3">
                        <div className="w-10 h-10 rounded-lg bg-amber-500/20 border border-amber-400/30 flex items-center justify-center">
                          <Rocket className="w-5 h-5 text-amber-400" />
                        </div>
                        <div>
                          <div className="text-sm font-semibold text-white">
                            {activeRocket.giftName}
                          </div>
                          <div className="text-xs text-slate-300 font-mono tabular-nums">
                            {activeRocket.senderName} · {formatCoins(activeRocket.totalCoins)} Coins ·{' '}
                            {economy.luckyPoolPct}% Lucky Pool ({formatCoins(activeRocket.luckyPoolTotal)})
                          </div>
                        </div>
                      </div>

                      <div className="text-right font-mono tabular-nums">
                        <div className="text-xs text-slate-300">
                          Claimants {activeRocket.claims.length} / {activeRocket.catcherLimit}
                        </div>
                        <div className="text-sm font-semibold text-amber-300">
                          {activeRocket.luckyPoolRemaining.toLocaleString()} coins left
                        </div>
                      </div>
                    </div>

                    {/* Progress bar for claims */}
                    <div className="w-full h-2 bg-slate-800 rounded-full overflow-hidden">
                      <div
                        className="h-full bg-amber-400 transition-transform duration-300 origin-left"
                        style={{
                          transform: `scaleX(${Math.min(
                            1,
                            activeRocket.claims.length / activeRocket.catcherLimit
                          )})`,
                        }}
                      />
                    </div>

                    <div className="flex flex-wrap items-center justify-between gap-3">
                      <div className="text-xs text-slate-300 font-mono tabular-nums">
                        Seat SP Split ({economy.seatSpSplitPct}%):{' '}
                        {activeRocket.seatSpAllocations
                          .map((amt, i) => `S${i + 1} +${amt.toLocaleString()}`)
                          .join(' · ')}
                      </div>

                      <div className="flex items-center gap-2">
                        <button
                          type="button"
                          disabled={
                            hasUserClaimedActive ||
                            activeRocket.claims.length >= activeRocket.catcherLimit
                          }
                          onClick={onClaimLuckyDrop}
                          className={`px-4 py-2 text-xs font-semibold rounded-lg transition-colors whitespace-nowrap shrink-0 ${
                            hasUserClaimedActive ||
                            activeRocket.claims.length >= activeRocket.catcherLimit
                              ? 'bg-slate-800 text-slate-400 cursor-not-allowed'
                              : 'bg-amber-500 hover:bg-amber-400 text-slate-950 cursor-pointer'
                          }`}
                        >
                          {hasUserClaimedActive
                            ? 'Claimed by You'
                            : activeRocket.claims.length >= activeRocket.catcherLimit
                              ? 'Pool Claimed'
                              : 'Catch Lucky Drop'}
                        </button>

                        {activeRocket.claims.length < activeRocket.catcherLimit && (
                          <button
                            type="button"
                            onClick={onSimulateRemainingClaims}
                            className="px-3.5 py-2 text-xs font-medium bg-white/10 hover:bg-white/20 text-white rounded-lg border border-white/15 transition-colors whitespace-nowrap shrink-0 cursor-pointer"
                          >
                            Fill {activeRocket.catcherLimit - activeRocket.claims.length} Slots
                          </button>
                        )}
                      </div>
                    </div>
                  </div>
                ) : (
                  <div className="bg-slate-900/70 border border-white/10 rounded-xl p-5 flex flex-wrap items-center justify-between gap-4">
                    <div>
                      <div className="text-sm font-semibold text-white">
                        Rocket &amp; Lucky Pool Stage Ready
                      </div>
                      <p className="text-xs text-slate-300 mt-1">
                        Launch any Rocket or Multi-Tier Lucky Back below to distribute {economy.luckyPoolPct}% to 10–50 catchers and split {economy.seatSpSplitPct}% SP across all 3 Boat Seats.
                      </p>
                    </div>
                    <div className="text-right font-mono tabular-nums">
                      <div className="text-xs text-slate-400">Host 40% Coin-Back Total</div>
                      <div className="text-base font-semibold text-emerald-400">
                        {room.totalReceiverCoinBack.toLocaleString()} coins
                      </div>
                    </div>
                  </div>
                )}
              </div>

              {/* Bottom 3-Seat SP Stage */}
              <div className="relative z-10 grid grid-cols-1 sm:grid-cols-3 gap-3">
                {room.seats.map((seat) => {
                  const isCurrentUser = seat.occupantUid === user.uid;
                  const seatHidden =
                    isCurrentUser && user.hideIdEnabled && userCanHide;
                  return (
                    <div
                      key={seat.seatNumber}
                      className="flex items-center justify-between gap-3 p-3 rounded-lg bg-slate-900/85 border border-white/15"
                    >
                      <div className="flex items-center gap-2.5 min-w-0">
                        <div className="w-9 h-9 rounded-full overflow-hidden bg-slate-800 border border-white/20 shrink-0 flex items-center justify-center text-xs font-semibold text-slate-200">
                          {seat.avatarUrl ? (
                            <img
                              src={seat.avatarUrl}
                              alt={seat.occupantName || `Seat ${seat.seatNumber}`}
                              referrerPolicy="no-referrer"
                              className="w-full h-full object-cover"
                            />
                          ) : (
                            <span>S{seat.seatNumber}</span>
                          )}
                        </div>
                        <div className="min-w-0">
                          <div className="text-xs font-semibold text-white truncate">
                            Seat {seat.seatNumber} · {seat.occupantName || 'Open Seat'}
                          </div>
                          <div className="text-[11px] font-mono tabular-nums text-slate-300 truncate">
                            {seat.occupantUid
                              ? `${formatUidDisplay(seat.occupantUid, seatHidden)} · `
                              : ''}
                            {seat.spEarned.toLocaleString()} SP
                          </div>
                        </div>
                      </div>

                      <button
                        type="button"
                        onClick={() => onToggleSeatMute(seat.seatNumber)}
                        aria-label={
                          seat.isMuted
                            ? `Unmute Seat ${seat.seatNumber}`
                            : `Mute Seat ${seat.seatNumber}`
                        }
                        className="p-2 rounded-md bg-white/5 hover:bg-white/15 text-slate-200 transition-colors shrink-0 cursor-pointer"
                      >
                        {seat.isMuted ? (
                          <MicOff className="w-3.5 h-3.5 text-red-400" />
                        ) : (
                          <Mic className="w-3.5 h-3.5 text-emerald-400" />
                        )}
                      </button>
                    </div>
                  );
                })}
              </div>
            </div>
          </div>

          {/* Rocket & Lucky Back Launcher + Exact Economy Accounting */}
          <div className="bg-white border border-slate-200 rounded-xl p-6 space-y-5">
            <div className="flex flex-wrap items-center justify-between gap-4">
              <div>
                <h3 className="font-display text-base font-semibold text-slate-900">
                  Rocket Launch &amp; Multi-Tier Lucky Back Console
                </h3>
                <p className="text-xs text-slate-600 mt-0.5">
                  Configure catcher limits ({economy.minCatcherLimit}–{economy.maxCatcherLimit} claimants) and preview the real-time split before launching.
                </p>
              </div>

              <button
                type="button"
                disabled={!userCanHide}
                onClick={onToggleUserHideId}
                className={`inline-flex items-center gap-1.5 px-3 py-1.5 text-xs font-medium rounded-lg border transition-colors whitespace-nowrap shrink-0 ${
                  userCanHide
                    ? 'border-slate-200 hover:bg-slate-50 text-slate-700 cursor-pointer'
                    : 'border-slate-100 bg-slate-50 text-slate-400 cursor-not-allowed'
                }`}
              >
                {user.hideIdEnabled && userCanHide ? (
                  <EyeOff className="w-3.5 h-3.5 text-blue-600" />
                ) : (
                  <Eye className="w-3.5 h-3.5" />
                )}
                {userCanHide
                  ? user.hideIdEnabled
                    ? `VIP ${user.vipLevel} Hide ID Active`
                    : `Enable VIP ${user.vipLevel} Hide ID`
                  : 'Hide ID unlocks at VIP 5'}
              </button>
            </div>

            {/* Gift Selector Strip */}
            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-6 gap-2.5">
              {featuredRockets.map((gift) => {
                const isSelected = gift.id === selectedGift.id;
                return (
                  <button
                    key={gift.id}
                    type="button"
                    onClick={() => handleGiftSelect(gift)}
                    className={`p-3 rounded-lg border text-left transition-colors cursor-pointer ${
                      isSelected
                        ? 'border-blue-600 bg-blue-50/70'
                        : 'border-slate-200 hover:border-slate-300 bg-white'
                    }`}
                  >
                    <div className="text-xs font-semibold text-slate-900 truncate">
                      {gift.name}
                    </div>
                    <div className="text-xs font-mono tabular-nums text-slate-600 mt-1">
                      {formatCoins(gift.coinCost)} coins
                    </div>
                    <div className="text-[11px] text-slate-500 mt-0.5">
                      {gift.category}
                    </div>
                  </button>
                );
              })}
            </div>

            {/* Catcher Limit + Launch CTA */}
            <div className="flex flex-wrap items-center justify-between gap-4 pt-2 border-t border-slate-100">
              <div className="flex flex-wrap items-center gap-4">
                <label className="text-xs font-medium text-slate-700 flex items-center gap-3">
                  <span>Catcher Limit ({economy.minCatcherLimit}–{economy.maxCatcherLimit})</span>
                  <input
                    type="range"
                    min={economy.minCatcherLimit}
                    max={economy.maxCatcherLimit}
                    value={catcherLimit}
                    onChange={(e) => setCatcherLimit(Number(e.target.value))}
                    className="w-36 accent-blue-600 cursor-pointer"
                  />
                  <span className="font-mono tabular-nums font-semibold text-slate-900 w-16">
                    {catcherLimit} slots
                  </span>
                </label>
              </div>

              <button
                type="button"
                disabled={user.coinBalance < selectedGift.coinCost}
                onClick={() => onLaunchGift(selectedGift, catcherLimit)}
                className={`inline-flex items-center gap-2 px-5 py-2.5 text-xs font-semibold rounded-lg transition-colors whitespace-nowrap shrink-0 ${
                  user.coinBalance >= selectedGift.coinCost
                    ? 'bg-blue-600 hover:bg-blue-700 text-white cursor-pointer'
                    : 'bg-slate-200 text-slate-500 cursor-not-allowed'
                }`}
              >
                <Rocket className="w-4 h-4" />
                Launch {selectedGift.name} ({formatCoins(selectedGift.coinCost)})
              </button>
            </div>

            {/* Real-Time Split Breakdown Table */}
            <div className="grid grid-cols-2 sm:grid-cols-5 gap-3 pt-2 border-t border-slate-100">
              <div>
                <div className="text-xs text-slate-500">
                  Receiver Coin-Back ({economy.receiverCoinBackPct}%)
                </div>
                <div className="text-sm font-mono tabular-nums font-semibold text-emerald-700 mt-0.5">
                  {previewSplit.receiverCoinBack.toLocaleString()}
                </div>
              </div>
              <div>
                <div className="text-xs text-slate-500">
                  Platform Comm. ({economy.platformCommissionPct}%)
                </div>
                <div className="text-sm font-mono tabular-nums font-semibold text-slate-900 mt-0.5">
                  {previewSplit.platformCommission.toLocaleString()}
                </div>
              </div>
              <div>
                <div className="text-xs text-slate-500">
                  Lucky Pool ({economy.luckyPoolPct}% / {catcherLimit})
                </div>
                <div className="text-sm font-mono tabular-nums font-semibold text-amber-700 mt-0.5">
                  {previewSplit.luckyPoolTotal.toLocaleString()}
                </div>
              </div>
              <div>
                <div className="text-xs text-slate-500">
                  3-Seat SP ({economy.seatSpSplitPct}% Total)
                </div>
                <div className="text-sm font-mono tabular-nums font-semibold text-blue-700 mt-0.5">
                  {previewSplit.seatSpTotal.toLocaleString()}
                </div>
              </div>
              <div>
                <div className="text-xs text-slate-500">Ecosystem Reserve (38%)</div>
                <div className="text-sm font-mono tabular-nums font-semibold text-slate-700 mt-0.5">
                  {previewSplit.systemReserve.toLocaleString()}
                </div>
              </div>
            </div>
          </div>
        </div>

        {/* Right Column: Live Room Chat + Recent Rocket Launches */}
        <div className="lg:col-span-4 flex flex-col gap-6">
          {/* Room Chat */}
          <div className="bg-white border border-slate-200 rounded-xl flex flex-col h-[430px]">
            <div className="px-4 py-3 border-b border-slate-200 flex items-center justify-between">
              <h3 className="font-display text-sm font-semibold text-slate-900">
                Room Feed &amp; Gift Ledger
              </h3>
              <span className="text-xs font-mono tabular-nums text-slate-500">
                {room.chatMessages.length} events
              </span>
            </div>

            <div className="flex-1 overflow-y-auto p-4 space-y-3">
              {room.chatMessages.map((msg) => (
                <div key={msg.id} className="text-xs leading-relaxed">
                  <div className="flex items-center gap-1.5 text-slate-500 font-mono tabular-nums">
                    <span>{msg.timestamp}</span>
                    <span aria-hidden="true">·</span>
                    <span
                      className={`font-sans font-semibold ${
                        msg.type === 'gift'
                          ? 'text-amber-700'
                          : msg.type === 'system'
                            ? 'text-blue-700'
                            : 'text-slate-800'
                      }`}
                    >
                      {msg.senderName}
                    </span>
                    {msg.senderUid !== '000000' && (
                      <>
                        <span aria-hidden="true">·</span>
                        <span>{formatUidDisplay(msg.senderUid, msg.isMasked)}</span>
                      </>
                    )}
                  </div>
                  <p className="text-slate-700 mt-0.5">{msg.text}</p>
                </div>
              ))}
            </div>

            <form
              onSubmit={handleSendChatSubmit}
              className="p-3 border-t border-slate-200 flex items-center gap-2"
            >
              <input
                type="text"
                value={chatInput}
                onChange={(e) => setChatInput(e.target.value)}
                placeholder="Send room message..."
                className="flex-1 px-3 py-2 text-xs bg-slate-50 border border-slate-200 rounded-lg focus:outline-none focus:border-blue-600"
              />
              <button
                type="submit"
                aria-label="Send message"
                className="p-2 bg-slate-900 hover:bg-slate-800 text-white rounded-lg transition-colors cursor-pointer"
              >
                <Send className="w-3.5 h-3.5" />
              </button>
            </form>
          </div>

          {/* Recent Rocket & Lucky Pool Claims Ledger */}
          <div className="bg-white border border-slate-200 rounded-xl p-5 flex-1 flex flex-col">
            <h3 className="font-display text-sm font-semibold text-slate-900">
              Recent Rocket Launches
            </h3>
            <p className="text-xs text-slate-500 mt-0.5">
              Verified splits across receiver, platform, lucky pool, and seats.
            </p>

            {launchHistory.length === 0 ? (
              <div className="flex-1 flex items-center justify-center text-xs text-slate-500 py-8">
                No rockets launched in this session yet.
              </div>
            ) : (
              <div className="mt-4 divide-y divide-slate-100">
                {launchHistory.slice(0, 5).map((item) => (
                  <div key={item.id} className="py-2.5 flex items-center justify-between gap-3">
                    <div className="min-w-0">
                      <div className="text-xs font-semibold text-slate-900 truncate">
                        {item.giftName}
                      </div>
                      <div className="text-[11px] font-mono tabular-nums text-slate-500">
                        {item.launchedAt} · {item.claims.length}/{item.catcherLimit} claims
                      </div>
                    </div>
                    <div className="text-right font-mono tabular-nums shrink-0">
                      <div className="text-xs font-semibold text-slate-900">
                        {formatCoins(item.totalCoins)}
                      </div>
                      <div className="text-[11px] text-emerald-700">
                        +{formatCoins(item.receiverCoinBack)} host
                      </div>
                    </div>
                  </div>
                ))}
              </div>
            )}
          </div>
        </div>
      </div>

      {/* 1-on-1 Video Call Modal (Unlocked on Follow) */}
      {videoCallOpen && (
        <div
          role="dialog"
          aria-modal="true"
          aria-label="1-on-1 Video Call"
          className="fixed inset-0 z-50 bg-slate-950/75 flex items-center justify-center p-4"
        >
          <div className="bg-slate-900 border border-white/15 rounded-xl w-full max-w-lg overflow-hidden text-white">
            <div className="relative h-64 bg-slate-950">
              <img
                src={room.hostAvatarUrl}
                alt={room.hostName}
                referrerPolicy="no-referrer"
                className={`w-full h-full object-cover ${
                  callCameraOff ? 'opacity-20' : 'opacity-90'
                }`}
              />
              <div className="absolute inset-0 bg-gradient-to-t from-slate-950/90 via-transparent to-slate-950/40" />
              <div className="absolute top-4 left-4 text-xs font-mono tabular-nums text-emerald-400 flex items-center gap-1.5">
                <PhoneCall className="w-3.5 h-3.5" />
                Encrypted 1-on-1 Video Call · Friendship Lv.{user.friendshipLevel}
              </div>
              <div className="absolute bottom-4 left-4">
                <div className="font-display text-base font-semibold">{room.hostName}</div>
                <div className="text-xs text-slate-300 font-mono">
                  {formatUidDisplay(room.hostUid, room.ownerHideId)} · Follow Verified
                </div>
              </div>
            </div>

            <div className="p-4 flex items-center justify-between gap-3 bg-slate-900">
              <div className="flex items-center gap-2">
                <button
                  type="button"
                  onClick={() => setCallMuted(!callMuted)}
                  className="px-3.5 py-2 text-xs font-medium rounded-lg bg-white/10 hover:bg-white/20 flex items-center gap-1.5 cursor-pointer"
                >
                  {callMuted ? <MicOff className="w-3.5 h-3.5 text-red-400" /> : <Mic className="w-3.5 h-3.5" />}
                  {callMuted ? 'Muted' : 'Mic On'}
                </button>
                <button
                  type="button"
                  onClick={() => setCallCameraOff(!callCameraOff)}
                  className="px-3.5 py-2 text-xs font-medium rounded-lg bg-white/10 hover:bg-white/20 flex items-center gap-1.5 cursor-pointer"
                >
                  {callCameraOff ? <VideoOff className="w-3.5 h-3.5 text-red-400" /> : <Video className="w-3.5 h-3.5" />}
                  {callCameraOff ? 'Cam Off' : 'Cam On'}
                </button>
              </div>

              <button
                type="button"
                onClick={() => setVideoCallOpen(false)}
                className="px-4 py-2 text-xs font-semibold rounded-lg bg-red-600 hover:bg-red-500 text-white flex items-center gap-1.5 cursor-pointer"
              >
                <PhoneOff className="w-3.5 h-3.5" />
                End Call
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};
