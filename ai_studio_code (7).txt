// 100 Gift Tiers (including Rockets and Multi-Tier Lucky Backs up to 20M coins)
const GIFT_PREFIXES = [
  'Harbor', 'Signal', 'Compass', 'Lantern', 'Anchor', 'Beacon', 'Vanguard', 'Atlas', 'Zenith', 'Sovereign',
];
const GIFT_NOUNS = [
  'Spark', 'Wave', 'Flare', 'Helm', 'Crest', 'Trident', 'Galleon', 'Cruiser', 'Rocket', 'Lucky Back',
];

export const GIFT_TIERS: GiftTier[] = Array.from({ length: 100 }, (_, idx) => {
  const tier = idx + 1;
  let coinCost = 10;
  if (tier <= 20) coinCost = tier * 25;
  else if (tier <= 40) coinCost = 500 + (tier - 20) * 475;
  else if (tier <= 60) coinCost = 10000 + (tier - 40) * 4500;
  else if (tier <= 80) coinCost = 100000 + (tier - 60) * 45000;
  else if (tier < 95) coinCost = 1000000 + (tier - 80) * 500000;
  else {
    const megaLadder = [10000000, 12000000, 14000000, 16000000, 18000000, 20000000];
    coinCost = megaLadder[tier - 95];
  }

  const isLuckyBack = tier >= 85 || tier % 10 === 0;
  const isRocket = tier >= 50 && (tier % 5 === 0 || tier >= 90);
  const category = isLuckyBack ? 'LuckyBack' : isRocket ? 'Rocket' : 'Standard';

  const prefix = GIFT_PREFIXES[idx % GIFT_PREFIXES.length];
  const noun = isLuckyBack
    ? 'Lucky Back'
    : isRocket
      ? 'Launch Rocket'
      : GIFT_NOUNS[Math.floor(idx / 10) % GIFT_NOUNS.length];

  const defaultCatcherLimit = tier >= 95 ? 50 : tier >= 80 ? 35 : tier >= 50 ? 25 : 10;

  return {
    id: `gift-tier-${tier}`,
    tier,
    name: `T${tier} ${prefix} ${noun}`,
    category,
    coinCost,
    isRocket,
    isLuckyBack,
    defaultCatcherLimit,
    xpYield: Math.round(coinCost * 0.5),
    spYield: Math.round(coinCost * 0.02),
  };
});
